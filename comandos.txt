1. MANUTENÇÃO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Atualizar yt-dlp
yt-dlp -U

# Ver versão instalada
yt-dlp --version

# Verificar formatos disponíveis para uma URL
yt-dlp -F --cookies cookies.txt URL

# Informações do vídeo sem baixar
yt-dlp --dump-json --cookies cookies.txt URL | python -m json.tool

2. DOWNLOAD DE ÁUDIO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

── OPUS (Melhor qualidade, menor tamanho) ──

yt-dlp \
  -f bestaudio \
  --extract-audio \
  --audio-format opus \
  --audio-quality 0 \
  --cookies cookies.txt \
  --embed-thumbnail \
  --embed-metadata \
  --add-metadata \
  --parse-metadata "title:%(title)s" \
  --parse-metadata "uploader:%(artist)s" \
  --sponsorblock-remove all \
  --no-playlist \
  -o "%(artist)s - %(title)s.%(ext)s" \
  --batch-file lista.txt

── MP3 (Máxima compatibilidade) ──

yt-dlp \
  -f bestaudio \
  --extract-audio \
  --audio-format mp3 \
  --audio-quality 0 \
  --cookies cookies.txt \
  --embed-thumbnail \
  --embed-metadata \
  --add-metadata \
  --sponsorblock-remove all \
  -o "%(artist)s - %(title)s.%(ext)s" \
  --batch-file lista.txt

── FLAC (Lossless — maior qualidade e tamanho) ──

yt-dlp \
  -f bestaudio \
  --extract-audio \
  --audio-format flac \
  --cookies cookies.txt \
  --embed-thumbnail \
  --embed-metadata \
  -o "%(artist)s - %(title)s.%(ext)s" \
  --batch-file lista.txt

── SIMPLIFICADO (rápido) ──

yt-dlp -x --audio-format opus --cookies cookies.txt -a lista.txt

3. DOWNLOAD DE VÍDEO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

── MELHOR QUALIDADE (sem limite de resolução) ──

yt-dlp \
  -f "bestvideo[ext=mp4]+bestaudio[ext=m4a]/best[ext=mp4]/best" \
  --merge-output-format mp4 \
  --cookies cookies.txt \
  --embed-thumbnail \
  --embed-metadata \
  --embed-subs \
  --sub-langs "pt,pt-BR,en" \
  --convert-subs srt \
  --embed-chapters \
  --sponsorblock-remove all \
  -o "%(uploader)s - %(title)s.%(ext)s" \
  --batch-file lista.txt

── 4K / 2160p ──

yt-dlp \
  -f "bestvideo[height<=2160][ext=mp4]+bestaudio[ext=m4a]/best" \
  --merge-output-format mp4 \
  --cookies cookies.txt \
  --embed-subs --sub-langs "pt,pt-BR,en" \
  -o "%(uploader)s - %(title)s [4K].%(ext)s" \
  --batch-file lista.txt

── 1080p Full HD ──

yt-dlp \
  -f "bestvideo[height<=1080][ext=mp4]+bestaudio[ext=m4a]/best[height<=1080][ext=mp4]/best" \
  --merge-output-format mp4 \
  --cookies cookies.txt \
  --embed-thumbnail --embed-metadata \
  --embed-subs --sub-langs "pt,pt-BR,en" \
  --sponsorblock-remove all \
  -o "%(uploader)s - %(title)s.%(ext)s" \
  --batch-file lista.txt

── 720p (HD, menor tamanho) ──

yt-dlp -f "bestvideo[height<=720]+bestaudio" \
  --merge-output-format mp4 --cookies cookies.txt -a lista.txt

── COM LIMITE DE VELOCIDADE (5 MB/s) ──

yt-dlp -f "bv+ba/b" --merge-output-format mp4 \
  --limit-rate 5M --cookies cookies.txt -a lista.txt

── DOWNLOAD ACELERADO COM aria2c (até 16 conexões) ──
# Requer: aria2 instalado (https://aria2.github.io)

yt-dlp -f "bv+ba/b" --merge-output-format mp4 \
  --downloader aria2c \
  --downloader-args "aria2c:-x 16 -s 16 -k 2M" \
  --cookies cookies.txt -a lista.txt

4. PLAYLISTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Playlist completa em pastas organizadas por playlist
yt-dlp -f "bv+ba/b" --merge-output-format mp4 \
  --cookies cookies.txt \
  -o "%(playlist_title)s/%(playlist_index)s - %(title)s.%(ext)s" \
  "URL_DA_PLAYLIST"

# Apenas áudio de playlist inteira
yt-dlp -x --audio-format opus --cookies cookies.txt \
  -o "%(playlist_title)s/%(playlist_index)s - %(title)s.%(ext)s" \
  "URL_DA_PLAYLIST"

# Baixar a partir de um índice específico (ex: a partir do vídeo 5)
yt-dlp --playlist-start 5 -f "bv+ba/b" --merge-output-format mp4 \
  --cookies cookies.txt "URL_DA_PLAYLIST"

# Intervalo específico da playlist (do vídeo 3 ao 10)
yt-dlp --playlist-items 3-10 -f "bv+ba/b" --merge-output-format mp4 \
  --cookies cookies.txt "URL_DA_PLAYLIST"

5. LEGENDAS E ACESSIBILIDADE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Baixar legendas automáticas (auto-generated) em PT
yt-dlp --write-auto-subs --sub-langs "pt,pt-BR" \
  --skip-download --cookies cookies.txt "URL"

# Baixar legendas manuais + automáticas
yt-dlp --write-subs --write-auto-subs \
  --sub-langs "pt,pt-BR,en" --convert-subs srt \
  --skip-download --cookies cookies.txt "URL"

# Incorporar legendas no vídeo (hardcode — permanente)
yt-dlp -f "bv+ba/b" --merge-output-format mp4 \
  --write-subs --embed-subs --sub-langs "pt" \
  --cookies cookies.txt "URL"

6. SPONSORBLOCK (Remover intros, ads, segmentos)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Remover TODOS os segmentos marcados pela comunidade
--sponsorblock-remove all

# Remover apenas patrocínios e intros
--sponsorblock-remove sponsor,intro

# Segmentos disponíveis:
#   sponsor      — patrocínios
#   intro        — introduções
#   outro        — encerramentos
#   selfpromo    — auto-promoção
#   interaction  — pedidos de like/sub
#   preview      — prévia do conteúdo
#   music_offtopic — partes sem música (em clipes musicais)

7. ARQUIVOS DE CONFIGURAÇÃO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

── Criar arquivo de configuração (evita repetir flags) ──
# Windows: %APPDATA%\yt-dlp\config
# Linux/macOS: ~/.config/yt-dlp/config

Exemplo ~/.config/yt-dlp/config:
-------------------------------
--cookies ~/cookies.txt
--embed-metadata
--embed-thumbnail
--embed-chapters
--sponsorblock-remove all
--merge-output-format mp4
-o "%(uploader)s - %(title)s.%(ext)s"
-------------------------------

── Como obter cookies.txt ──

MÉTODO 1 — Extensão (Recomendado):
  1. Instale "Get cookies.txt LOCALLY" no navegador
  2. Acesse o YouTube logado
  3. Clique na extensão → Export → salve como cookies.txt

MÉTODO 2 — Via yt-dlp direto:
  yt-dlp --cookies-from-browser chrome   (ou firefox, edge, brave)

── Criar lista.txt ──
  Um URL por linha:
  https://www.youtube.com/watch?v=XXXXX
  https://www.youtube.com/watch?v=YYYYY

8. RETOMAR DOWNLOADS INTERROMPIDOS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Usar arquivo de histórico para não re-baixar o que já foi feito
yt-dlp --download-archive historico.txt \
  -f "bv+ba/b" --merge-output-format mp4 \
  --cookies cookies.txt -a lista.txt

# Continuar download parcial (fragmentos incompletos)
yt-dlp --continue --cookies cookies.txt "URL"

9. NOMES DE ARQUIVO — TEMPLATES %(variável)s
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  %(title)s          Título do vídeo
  %(uploader)s       Nome do canal
  %(artist)s         Artista (metadados)
  %(album)s          Álbum
  %(upload_date)s    Data de upload (YYYYMMDD)
  %(id)s             ID do vídeo
  %(ext)s            Extensão do arquivo
  %(playlist_title)s Nome da playlist
  %(playlist_index)s Índice na playlist
  %(resolution)s     Resolução (ex: 1920x1080)
  %(duration_string)s Duração (ex: 10:30)

  Exemplo organizado por data:
  -o "%(upload_date>%Y-%m)s/%(uploader)s - %(title)s.%(ext)s"

10. EXEMPLOS RÁPIDOS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

yt-dlp -F --cookies cookies.txt "URL"

yt-dlp --write-thumbnail --skip-download --cookies cookies.txt "URL"

# Baixar apenas metadados JSON
yt-dlp --dump-json --cookies cookies.txt "URL" > info.json

# Simular download sem baixar (teste)
yt-dlp --simulate --cookies cookies.txt "URL"
